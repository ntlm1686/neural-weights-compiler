def build_policy():
    import numpy as np

    # ------------------------------------------------------------------
    # Hand-written weights for 600 -> 32 -> 32 -> 6
    #
    # Observation layout:
    #   index = (y * 10 + x) * 6 + channel
    # channels:
    #   0 = player
    #   1 = enemy
    #   2 = gold
    #   4 = trail
    #
    # Policy idea:
    #   - Detect enemy rows and player row.
    #   - If an enemy is in the player's row, move vertically away.
    #   - If an enemy is above/below, mostly stay put unless it is in
    #     the same row or the player is on the near side.
    #   - Small horizontal centering bias.
    #   - Avoid fire because it has no effect in the provided env.
    # ------------------------------------------------------------------

    def idx(y, x, c):
        return (y * 10 + x) * 6 + c

    W1 = np.zeros((32, 600), dtype=np.float32)
    b1 = np.zeros(32, dtype=np.float32)

    # Row 0: player_y centered, optional feature.
    for y in range(10):
        for x in range(10):

    for y in range(10):

            W1[1, idx(y, x, 1)] = 0.2

    for y in range(10):
        for x in range(10):
            W1[2, idx(y, x, 2)] = 1.0

    # Row 3: trail count.
    for y in range(10):
        for x in range(10):
            W1[3, idx(y, x, 4)] = 0.2

    for k in range(8):

        for x in range(10):

    for k in range(4):

    #   enemy_above[k] = tanh(enemy_row[k] + sum(player_row[p] for p > k) - 1)
    for k in range(8):

    W2[27, 11] = 1.0   # enemy at y=1
