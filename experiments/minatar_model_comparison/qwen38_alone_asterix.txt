class ExecutableMLP:
    def __init__(self, weights, biases, activations):
        self.weights = weights
        self.biases = biases
        self.activations = activations
    
    def forward(self, x):
        for i in range(len(self.weights)):
            x = x @ self.weights[i] + self.biases[i]
            if i < len(self.weights) - 1:
                x = np.tanh(x)
        return x
