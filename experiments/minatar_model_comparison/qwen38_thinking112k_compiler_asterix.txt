def build_policy():
    def cell(r, c, ch):
        return (r * 10 + c) * 6 + ch

    def row_cells(r, ch):
        return [cell(r, c, ch) for c in range(10)]

    def rect_cells(r0, r1, c0, c1, ch):
        out = []
        for r in range(r0, r1 + 1):
            for c in range(c0, c1 + 1):
                out.append(cell(r, c, ch))
        return out

    prog = MLPProgram(
        input_dim=600,
        h1_dim=32,
        h2_dim=32,
        output_dim=6,
        activation="tanh",
    )

    # H1 relative enemy row detectors
    for r in range(2, 9):
        prog.add_h1_and_gate(f"above_{r}", row_cells(r, 0), row_cells(r - 1, 1), gain=20.0)
    for r in range(1, 8):
        prog.add_h1_and_gate(f"below_{r}", row_cells(r, 0), row_cells(r + 1, 1), gain=20.0)
    for r in range(1, 9):
        prog.add_h1_and_gate(f"same_{r}", row_cells(r, 0), row_cells(r, 1), gain=20.0)

    # H1 global half detectors
    prog.add_h1_threshold("player_top", {i: 1.0 for i in rect_cells(1, 2, 0, 9, 0)}, threshold=0.5, gain=20.0)
    prog.add_h1_threshold("player_bottom", {i: 1.0 for i in rect_cells(7, 8, 0, 9, 0)}, threshold=0.5, gain=20.0)
    prog.add_h1_threshold("enemy_left", {i: 1.0 for i in rect_cells(1, 8, 0, 4, 1)}, threshold=0.5, gain=20.0)
    prog.add_h1_threshold("enemy_right", {i: 1.0 for i in rect_cells(1, 8, 5, 9, 1)}, threshold=0.5, gain=20.0)
    prog.add_h1_threshold("enemy_top", {i: 1.0 for i in rect_cells(1, 4, 0, 9, 1)}, threshold=0.5, gain=20.0)
    prog.add_h1_threshold("enemy_bottom", {i: 1.0 for i in rect_cells(5, 8, 0, 9, 1)}, threshold=0.5, gain=20.0)
    prog.add_h1_threshold("gold_top", {i: 1.0 for i in rect_cells(1, 4, 0, 9, 2)}, threshold=0.5, gain=20.0)
    prog.add_h1_threshold("gold_bottom", {i: 1.0 for i in rect_cells(5, 8, 0, 9, 2)}, threshold=0.5, gain=20.0)
    prog.add_h1_threshold("gold_left", {i: 1.0 for i in rect_cells(1, 8, 0, 4, 2)}, threshold=0.5, gain=20.0)
    prog.add_h1_threshold("gold_right", {i: 1.0 for i in rect_cells(1, 8, 5, 9, 2)}, threshold=0.5, gain=20.0)

    # H2 action scores
    down_terms = {f"above_{r}": 1.0 for r in range(2, 9)}
    down_terms["same_1"] = 1.0
    prog.add_h2_gate("h2_down", down_terms, bias=float(len(down_terms)))

    up_terms = {f"below_{r}": 1.0 for r in range(1, 8)}
    for r in range(2, 9):
        up_terms[f"same_{r}"] = 1.0
    prog.add_h2_gate("h2_up", up_terms, bias=float(len(up_terms)))

    prog.add_h2_gate("h2_left", {"enemy_right": 1.0}, bias=1.0)
    prog.add_h2_gate("h2_right", {"enemy_left": 1.0}, bias=1.0)

    prog.add_h2_gate("h2_gold_up", {"gold_top": 1.0, "enemy_top": -1.0}, bias=0.0)
    prog.add_h2_gate("h2_gold_down", {"gold_bottom": 1.0, "enemy_bottom": -1.0}, bias=0.0)
    prog.add_h2_gate("h2_gold_left", {"gold_left": 1.0, "enemy_left": -1.0}, bias=0.0)
    prog.add_h2_gate("h2_gold_right", {"gold_right": 1.0, "enemy_right": -1.0}, bias=0.0)

    prog.output_bias = [-10.0, -10.0, -10.0, -10.0, -10.0, -10.0]
    prog.set_output_action(0, {})
    prog.set_output_action(1, {"h2_left": 25.0, "h2_gold_left": 10.0})
    prog.set_output_action(2, {"h2_up": 32.0, "h2_gold_up": 10.0})

    prog.set_output_action(5, {})

    return MLPCompilerBackend.compile(prog)

def build_policy():
    def cell(r, c, ch):
        return (r * 10 + c) * 6 + ch

    def row_cells(r, ch):
        return [cell(r, c, ch) for c in range(10)]

    def rect_cells(r0, r1, c0, c1, ch):
        out = []
        for r in range(r0, r1 + 1):
            for c in range(c0, c1 + 1):
                out.append(cell(r, c, ch))
        return out

    prog = MLPProgram(
        input_dim=600,
        h1_dim=32,
        h2_dim=32,
        output_dim=6,
        activation="tanh",
    )

    # Layer 1: relative enemy-row detectors.
    # above_r: player in row r, enemy in row r-1.
    for r in range(2, 9):
        prog.add_h1_and_gate(
            f"above_{r}",
            row_cells(r, 0),
            row_cells(r - 1, 1),
            gain=20.0,
        )

    # below_r: player in row r, enemy in row r+1.
    for r in range(1, 8):
        prog.add_h1_and_gate(
            f"below_{r}",
            row_cells(r, 0),
            row_cells(r + 1, 1),
            gain=20.0,
        )

    # same_r: player and enemy in row r.
    for r in range(1, 9):
        prog.add_h1_and_gate(
            f"same_{r}",
            row_cells(r, 0),
            row_cells(r, 1),
            gain=20.0,
        )

    # Layer 1: coarse global detectors.
    prog.add_h1_threshold(
        "player_top",
        {i: 1.0 for i in rect_cells(1, 2, 0, 9, 0)},
        threshold=0.5,
        gain=20.0,
    )
    prog.add_h1_threshold(
        "player_bottom",
        {i: 1.0 for i in rect_cells(7, 8, 0, 9, 0)},
        threshold=0.5,
        gain=20.0,
    )

    prog.add_h1_threshold(
        "enemy_left",
        {i: 1.0 for i in rect_cells(1, 8, 0, 4, 1)},
        threshold=0.5,
        gain=20.0,
    )
    prog.add_h1_threshold(
        "enemy_right",
        {i: 1.0 for i in rect_cells(1, 8, 5, 9, 1)},
        threshold=0.5,
        gain=20.0,
    )
    prog.add_h1_threshold(
        "enemy_top",
        {i: 1.0 for i in rect_cells(1, 4, 0, 9, 1)},
        threshold=0.5,
        gain=20.0,
    )
    prog.add_h1_threshold(
        "enemy_bottom",
        {i: 1.0 for i in rect_cells(5, 8, 0, 9, 1)},
        threshold=0.5,
        gain=20.0,
    )

    prog.add_h1_threshold(
        "gold_top",
        {i: 1.0 for i in rect_cells(1, 4, 0, 9, 2)},
        threshold=0.5,
        gain=20.0,
    )
    prog.add_h1_threshold(
        "gold_bottom",
        {i: 1.0 for i in rect_cells(5, 8, 0, 9, 2)},
        threshold=0.5,
        gain=20.0,
    )
    prog.add_h1_threshold(
        "gold_left",
        {i: 1.0 for i in rect_cells(1, 8, 0, 4, 2)},
        threshold=0.5,
        gain=20.0,
    )
    prog.add_h1_threshold(
        "gold_right",
        {i: 1.0 for i in rect_cells(1, 8, 5, 9, 2)},
        threshold=0.5,
        gain=20.0,
    )

    # Layer 2: raw vertical action scores.
    down_terms = {f"above_{r}": 1.0 for r in range(2, 9)}
    down_terms["same_1"] = 1.0
    prog.add_h2_gate("h2_down", down_terms, bias=float(len(down_terms)))

    up_terms = {f"below_{r}": 1.0 for r in range(1, 8)}
    for r in range(2, 9):
        up_terms[f"same_{r}"] = 1.0
    prog.add_h2_gate("h2_up", up_terms, bias=float(len(up_terms)))

    # Layer 2: vertical conflict detectors.
    # If enemies are both above and below, vertical movement is unsafe.
    for r in range(2, 8):
        prog.add_h2_gate(
            f"conflict_{r}",
            {f"above_{r}": 10.0, f"below_{r}": 10.0},
            bias=-15.0,
        )

    # Top boundary conflict: row 1 with same-row and below-row enemies.
    prog.add_h2_gate(
        "top_conflict",
        {"same_1": 10.0, "below_1": 10.0},
        bias=-15.0,
    )

    # Bottom boundary conflict: row 8 with above-row and same-row enemies.
    prog.add_h2_gate(
        "bottom_conflict",
        {"above_8": 10.0, "same_8": 10.0},
        bias=-15.0,
    )

    # Layer 2: horizontal avoidance scores.
    prog.add_h2_gate("h2_left", {"enemy_right": 1.0}, bias=1.0)
    prog.add_h2_gate("h2_right", {"enemy_left": 1.0}, bias=1.0)

    # Layer 2: gold-seeking scores, suppressed by enemies in the same half.
    prog.add_h2_gate(
        "h2_gold_up",
        {"gold_top": 1.0, "enemy_top": -1.0},
        bias=0.0,
    )
    prog.add_h2_gate(
        "h2_gold_down",
        {"gold_bottom": 1.0, "enemy_bottom": -1.0},
        bias=0.0,
    )
    prog.add_h2_gate(
        "h2_gold_left",
        {"gold_left": 1.0, "enemy_left": -1.0},
        bias=0.0,
    )
    prog.add_h2_gate(
        "h2_gold_right",
        {"gold_right": 1.0, "enemy_right": -1.0},
        bias=0.0,
    )

    # Output scoring.
    # Conflict units are approximately -1 when inactive and +1 when active.
    # Negative output weights plus a compensating bias make inactive conflicts
    # contribute zero and active conflicts strongly suppress vertical actions.
    conflict_penalty = 50.0

    middle_conflict_terms = {
        f"conflict_{r}": -conflict_penalty for r in range(2, 8)
    }

    down_conflict_terms = dict(middle_conflict_terms)
    down_conflict_terms["top_conflict"] = -conflict_penalty

    up_conflict_terms = dict(middle_conflict_terms)
    up_conflict_terms["bottom_conflict"] = -conflict_penalty

    bias_down = -10.0 - conflict_penalty * len(down_conflict_terms)
    bias_up = -10.0 - conflict_penalty * len(up_conflict_terms)

    prog.set_output_action(0, {}, bias=-5.0)
    prog.set_output_action(
        1,
        {"h2_left": 25.0, "h2_gold_left": 12.0},
        bias=-10.0,
    )
    prog.set_output_action(
        2,
        {
            "h2_up": 32.0,
            "h2_gold_up": 12.0,
            **up_conflict_terms,
        },
        bias=bias_up,
    )
    prog.set_output_action(
        3,
        {"h2_right": 25.0, "h2_gold_right": 12.0},
        bias=-10.0,
    )
    prog.set_output_action(
        4,
        {
            "h2_down": 33.0,
            "h2_gold_down": 12.0,
            **down_conflict_terms,
        },
        bias=bias_down,
    )
    prog.set_output_action(5, {}, bias=-20.0)

    return MLPCompilerBackend.compile(prog)
