import numpy as np


def build_policy():
    """
    Return a hand-designed 600 -> 32 -> 32 -> 6 MLP policy.

    The policy uses only linear features of the observation:
      - player offset from center
      - enemy/gold/trail counts and center-relative sums
      - side counts for enemies/gold/trails

    It then forms simple action logits:
      - move away from the side with more enemies
      - if no strong enemy pressure, move toward gold
      - otherwise prefer no-op
    """
    W1 = np.zeros((600, 32), dtype=np.float32)
    b1 = np.zeros(32, dtype=np.float32)

    def idx(y, x, c):
        return (y * 10 + x) * 6 + c

    # Feature 0: player_x - 5
    # Feature 1: player_y - 5
    # Feature 2: enemy count
    # Feature 3: gold count
    # Feature 4: trail count
    # Feature 5: sum(enemy_x - 5)
    # Feature 6: sum(enemy_y - 5)
    # Feature 7: sum(gold_x - 5)
    # Feature 8: sum(gold_y - 5)
    # Feature 9: sum(trail_x - 5)
    # Feature 10: sum(trail_y - 5)
    # Features 11-22: side counts for enemy/gold/trail
    for y in range(10):
        for x in range(10):
            dx = float(x - 5)
            dy = float(y - 5)

            # player channel
            W1[idx(y, x, 0), 0] = dx
            W1[idx(y, x, 0), 1] = dy

            # enemy channel
            W1[idx(y, x, 1), 2] = 1.0
            W1[idx(y, x, 1), 5] = dx
            W1[idx(y, x, 1), 6] = dy
            if x < 5:
                W1[idx(y, x, 1), 11] = 1.0
            elif x > 5:
                W1[idx(y, x, 1), 12] = 1.0
            if y < 5:
                W1[idx(y, x, 1), 13] = 1.0
            elif y > 5:
                W1[idx(y, x, 1), 14] = 1.0

            # gold channel
            W1[idx(y, x, 2), 3] = 1.0
            W1[idx(y, x, 2), 7] = dx
            W1[idx(y, x, 2), 8] = dy
            if x < 5:
                W1[idx(y, x, 2), 15] = 1.0
            elif x > 5:
                W1[idx(y, x, 2), 16] = 1.0
            if y < 5:
                W1[idx(y, x, 2), 17] = 1.0
            elif y > 5:
                W1[idx(y, x, 2), 18] = 1.0

            # trail channel
            W1[idx(y, x, 4), 4] = 1.0
            W1[idx(y, x, 4), 9] = dx
            W1[idx(y, x, 4), 10] = dy
            if x < 5:
                W1[idx(y, x, 4), 19] = 1.0
            elif x > 5:
                W1[idx(y, x, 4), 20] = 1.0
            if y < 5:
                W1[idx(y, x, 4), 21] = 1.0
            elif y > 5:
                W1[idx(y, x, 4), 22] = 1.0

    # Second hidden layer: pass the first-layer features through unchanged.
    W2 = np.eye(32, dtype=np.float32)
    b2 = np.zeros(32, dtype=np.float32)

    # Third layer: map features to action logits.
    W3 = np.zeros((32, 6), dtype=np.float32)
    b3 = np.zeros(6, dtype=np.float32)

    # Action indices:
    # 0 = no-op, 1 = left, 2 = up, 3 = right, 4 = down, 5 = fire
    scale = 6.0

    # Enemy pressure: move away from the side with more enemy mass.
    # enemy_x feature positive means enemies are to the right -> prefer left.
    W3[5, 1] += scale
    W3[5, 3] -= scale

    # enemy_y feature positive means enemies are below -> prefer up.
    W3[6, 2] += scale
    W3[6, 4] -= scale

    # Side-count reinforcement.
    W3[11, 3] += 0.5 * scale  # enemies left -> move right
    W3[12, 1] += 0.5 * scale  # enemies right -> move left
    W3[13, 4] += 0.5 * scale  # enemies top -> move down
    W3[14, 2] += 0.5 * scale  # enemies bottom -> move up

    # Gold attraction: move toward gold mass.
    W3[7, 3] += 0.75 * scale  # gold right -> move right
    W3[7, 1] -= 0.75 * scale
    W3[8, 4] += 0.75 * scale  # gold bottom -> move down
    W3[8, 2] -= 0.75 * scale

    # Slight gold side-count reinforcement.
    W3[15, 1] += 0.25 * scale
    W3[16, 3] += 0.25 * scale
    W3[17, 2] += 0.25 * scale
    W3[18, 4] += 0.25 * scale

    # If no enemies and no gold, weakly return toward center.
    W3[0, 3] += 0.25 * scale  # player left of center -> move right
    W3[0, 1] -= 0.25 * scale
    W3[1, 4] += 0.25 * scale  # player above center -> move down
    W3[1, 2] -= 0.25 * scale

    # Prefer no-op when there is no strong directional signal.
    b3[0] = 2.0
    b3[5] = -5.0

    # If the harness provides ExecutableMLP, use it.  Otherwise provide a
    # minimal compatible callable MLP so build_policy remains self-contained.
    if "ExecutableMLP" in globals():
        mlp = ExecutableMLP
    else:
        class ExecutableMLP:
            def __init__(self, weights, biases):
                self.weights = weights
                self.biases = biases

            def __call__(self, x):
                x = np.asarray(x, dtype=np.float32).reshape(-1)
                for W, b in zip(self.weights, self.biases):
                    x = np.tanh(x @ W + b)
                return x

    return ExecutableMLP([W1, W2, W3], [b1, b2, b3])
